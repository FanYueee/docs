import { Callout } from 'fumadocs-ui/components/callout';
import { Step, Steps } from 'fumadocs-ui/components/steps';

<Steps>
<Step>
輸入對方的電子郵件，選擇權限：**僅檢視**、**操作員**或**全部權限**。想細分可展開「自訂權限」。
</Step>
<Step>
按「**新增伺服器使用者**」送出邀請。

<Shot src="/images/gamecloud/users.png" alt="使用者" />
</Step>
</Steps>

<Callout title="對方還沒註冊也可以邀請" type="info">
會顯示為「待接受邀請」，對方登入後自動生效。
</Callout>
