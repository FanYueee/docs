---
title: 單一外掛
description: 搜尋並一鍵安裝外掛（或模組）。
icon: Puzzle
---

import { Callout } from 'fumadocs-ui/components/callout';
import { Step, Steps } from 'fumadocs-ui/components/steps';

在「軟體 → 單一外掛」可以從 Modrinth、Hangar 或 CurseForge 搜尋外掛並一鍵安裝，需要的前置外掛會自動一起安裝。模組伺服器這裡會顯示「單一模組」。

{/* tutorial:start */}
<Steps>
<Step>

<Shot src="/images/tutorials/plugins/01.webp" alt="安裝單一外掛 步驟 1：在伺服器頁面上方點「軟體」，打開選單" width={2800} height={1800} caption="在伺服器頁面上方點「軟體」，打開選單。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/02.webp" alt="安裝單一外掛 步驟 2：選「單一外掛」" width={2800} height={1800} caption="選「單一外掛」。（模組伺服器這裡會顯示「單一模組」）" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/03.webp" alt="安裝單一外掛 步驟 3：上方卡片會顯示系統自動判斷的伺服器類型、Minecraft 版本和安裝位置" width={2800} height={1800} caption="上方卡片會顯示系統自動判斷的伺服器類型、Minecraft 版本和安裝位置。判斷錯誤時可以按「不對嗎？手動選擇」修正。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/04.webp" alt="安裝單一外掛 步驟 4：點「找新外掛」" width={2800} height={1800} caption="點「找新外掛」。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/05.webp" alt="安裝單一外掛 步驟 5：選擇下載來源：Modrinth、Hangar 或 CurseForge" width={2800} height={1800} caption="選擇下載來源：Modrinth、Hangar 或 CurseForge。這裡以 Modrinth 為例。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/06.webp" alt="安裝單一外掛 步驟 6：輸入外掛名稱，例如 LuckPerms，按 Enter 或「搜尋」" width={2800} height={1800} caption="輸入外掛名稱，例如 LuckPerms，按 Enter 或「搜尋」。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/07.webp" alt="安裝單一外掛 步驟 7：在搜尋結果中點選要安裝的外掛" width={2800} height={1800} caption="在搜尋結果中點選要安裝的外掛。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/08.webp" alt="安裝單一外掛 步驟 8：選擇要安裝的版本" width={2800} height={1800} caption="選擇要安裝的版本。每個版本會列出支援的 Minecraft 版本，請選擇符合伺服器版本（這台是 1.21.4）的那一個。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/09.webp" alt="安裝單一外掛 步驟 9：確認要安裝的檔案與大小" width={2800} height={1800} caption="確認要安裝的檔案與大小。需要的前置外掛會一起列出並自動安裝。沒問題就按確認安裝。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/10.webp" alt="安裝單一外掛 步驟 10：安裝進度會顯示在這裡" width={2800} height={1800} caption="安裝進度會顯示在這裡。完成後依提示重新啟動伺服器，外掛就會載入。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/11.webp" alt="安裝單一外掛 步驟 11：點「已安裝的外掛」可以看到伺服器上所有外掛" width={2800} height={1800} caption="點「已安裝的外掛」可以看到伺服器上所有外掛。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/12.webp" alt="安裝單一外掛 步驟 12：每個外掛右側可以「停用」（暫時不載入，檔案保留）或「移除」" width={2800} height={1800} caption="每個外掛右側可以「停用」（暫時不載入，檔案保留）或「移除」。有新版本時也會顯示「更新」。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/13.webp" alt="安裝單一外掛 步驟 13：要刪除外掛時按「移除」" width={2800} height={1800} caption="要刪除外掛時按「移除」。" />
</Step>
<Step>

<Shot src="/images/tutorials/plugins/14.webp" alt="安裝單一外掛 步驟 14：確認視窗會顯示要刪除的檔案" width={2800} height={1800} caption="確認視窗會顯示要刪除的檔案。只想暫時不用的話建議改用「停用」。確定刪除就按移除。" />
</Step>
</Steps>
